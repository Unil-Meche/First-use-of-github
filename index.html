highlightBox.visible = false; scene.add(highlightBox); const ISLAND_RADIUS = 12; // Compact, cozy survival island size const blockGeo = new THREE.BoxGeometry(1, 1, 1); const worldMap = new Map(); const meshesArray = []; function posKey(x, y, z) { return `${x},${y},${z}`; } function addBlock(x, y, z, typeId) { const key = posKey(x, y, z); if (worldMap.has(key)) return; const blockDef = BLOCK_TYPES.find(b => b.id === typeId); if (idx > -1) meshesArray.splice(idx, 1); worldMap.delete(key); } function spawnTree(tx, ty, tz) { // Trunk for (let y = 0; y < 4; y++) { addBlock(tx, ty + y, tz, 5); // Oak Log } // Leaves canopy for (let lx = -2; lx <= 2; lx++) { for (let lz = -2; lz <= 2; lz++) { for (let ly = 3; ly <= 5; ly++) { if (Math.abs(lx) === 2 && Math.abs(lz) === 2 && ly === 5) continue; if (lx === 0 && lz === 0 && ly < 5) continue; addBlock(tx + lx, ty + ly, tz + lz, 6); // Leaves } } } } function generateIsland() { for (const m of meshesArray) scene.remove(m); worldMap.clear(); meshesArray.length = 0; const seed = Math.random() * 1000; for (let x = -ISLAND_RADIUS; x <= ISLAND_RADIUS; x++) { for (let z = -ISLAND_RADIUS; z <= ISLAND_RADIUS; z++) { const distFromCenter = Math.hypot(x, z); if (distFromCenter > ISLAND_RADIUS) { // Surround with ocean water level for (let y = 0; y <= 1; y++) { addBlock(x, y, z, 8); // Water } continue; } // Island height curve (higher in center, sloping down to beach) const hNoise = Math.sin((x + seed) * 0.15) * Math.cos((z + seed) * 0.15) * 2; const surfaceY = Math.max(1, Math.floor(4 - (distFromCenter / ISLAND_RADIUS) * 3 + hNoise)); for (let y = 0; y <= surfaceY; y++) { let blockId = 3; // Stone base if (y === surfaceY) { blockId = surfaceY <= 1 ? 7 : 1; // Sand on beach, Grass on top } else if (y >= surfaceY - 2) { blockId = surfaceY <= 1 ? 7 : 2; // Dirt } addBlock(x, y, z, blockId); } // Spawn a tree or flower near center if (surfaceY >= 3 && distFromCenter < 6 && Math.random() < 0.08) { spawnTree(x, surfaceY + 1, z); } // ========================================== // PART 4: MINING, BLOCK HP & TOOL DAMAGE // ========================================== const raycaster = new THREE.Raycaster(); raycaster.far = 6.5; function getTargetBlock() { raycaster.setFromCamera(new THREE.Vector2(0, 0), camera);
